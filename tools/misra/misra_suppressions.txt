# Rule 15.5（関数の出口は末尾の1つのみとする）

#   実装でも一般的な逸脱。単一出口へ書き換えるとネストが深くなり可読性が
#   かえって下がるため、プロジェクト全体で逸脱とする。
misra-c2012-15.5

# Rule 8.7（1つの翻訳単位からしか参照されない関数・オブジェクトは内部結合とする）

#   （GetVersionInfo 等、現構成で他モジュールから呼ばれていないもの）。
#   仕様上の公開インタフェースであり static にはできないため逸脱とする。
misra-c2012-8.7

# Rule 11.8（Csm_MacVerify: const uint8* macPtr を Crypto_JobType.macPtr(uint8*)へ
#   渡す際に const を外している）
#   Crypto_ProcessJob() を実装まで追跡して確認済み: job->macPtr への書き込みは
#   CRYPTO_MACGENERATE 分岐でのみ発生し、CRYPTO_MACVERIFY 分岐（本呼び出しが
#   常に指定する service）では定数時間比較の読み取りにしか使われない。
#   Crypto_JobType.macPtr が生成/検証の両方で1つの非const フィールドを共用する
#   設計自体が根本原因だが、型を分離する構造体変更は影響範囲が広いため、
#   現時点ではコメントで安全性を明記した上で逸脱とする（2026-09 レビュー）。
misra-c2012-11.8:src/Bsw/Csm/Csm.c:237

# Rule 7.3（数値リテラルのサフィックスに小文字の l を使わない）
#   Cppcheck アドオンの実装（misra.py の misra_7_3）は AST ではなく生トークン
#   への単純な正規表現 `^(0[xX])?[0-9a-fA-FpP.]+[Uu]*l+[Uu]*$` でこのルールを
#   判定しており、トークンが本当に数値リテラルかどうかを検証していない。
#   'a'〜'f' は 16進数字としても有効な文字であるため、末尾が小文字 l で終わる
#   識別子（例: WdgM.c の変数名 `dl`）を誤って数値リテラルと誤認し、全4件が
#   誤検知（false positive）と確認済み（2026-09 レビュー、変数名を実際に
#   参照し数値リテラルが存在しないことを目視確認済み）。本プロジェクトの
#   数値リテラルは一貫して大文字の U/UL サフィックスを使う規約のため、本物の
#   小文字 l 使用が紛れ込むリスクは低いと判断し、ツールの制約としてルール
#   全体を逸脱とする。
misra-c2012-7.3

# Rule 2.5（未使用のマクロ定義を残さない）— インライン抑制の運用方針
#   本ファイルでは扱わず、該当 #define の直前に
#   `/* cppcheck-suppress misra-c2012-2.5 */` を置いて個別に抑制している。
#   抑制してよいのは次のいずれかに限る（2026-09 方針確定）:
#     (a) AUTOSAR SWS に記載がある Service ID / エラーコード / 戻り値。
#         仕様の表と一対一に保つため、現構成で参照されなくても削除しない
#         （Service ID は各 SWS の「Service ID[hex]」を実測して照合済み）。
#     (b) src/ からは見えないが test/ から参照されているもの。
#     (c) 設計ドキュメントとして機能している定数。選択肢の一覧表（CAN_BAUDRATE_*、
#         CAN_CRYSTAL_*）、Os タスク ID の対応表（BSWM_OS_TASK_*）、コメントが
#         設計値の出典として名指しで参照している定数（WDGM_HW_TRIGGER_CYCLE_MS 等）。
#         一部だけ削除すると表が歯抜けになり、コメントの参照先も失われるため残す。
#   上記以外の内部向けで未使用の #define は、抑制せず削除する。

# Rule 8.9（1つの関数からしか参照されないオブジェクトはブロックスコープで定義する）
#   インライン抑制（`/* cppcheck-suppress misra-c2012-8.9 */`）で個別に逸脱とする。
#   逸脱とするのは次のいずれかに限る（2026-09 方針確定）:
#     (a) ECUC の設定コンテナに相当する設定テーブル・設定用ステート
#         （Dio_PortConfig、Port_PinConfig、Dem_AgingThresholdTable、
#         Dcm_SidSessionTable、E2EXf_PBCfg.c の各テーブル/ステート）。
#         AUTOSAR では設定データはファイルスコープの const テーブルとして
#         生成される。E2EXf_PBCfg.c のステート変数はファイルスコープの設定
#         構造体の初期化子からアドレス参照されるため、ブロックスコープには
#         そもそも置けない（cppcheck は初期化子側の参照を数えない）。
#     (b) 呼び出しをまたいで保持する状態で、同種の状態変数群と並べて管理し
#         ているもの（NvM_ActiveCrc。MemIf_Write() がジョブ完了までポインタを
#         保持するため static 必須）。
#     (c) 本番ビルドでは1関数からしか見えないが、単体テスト用アクセサ
#         （*_UNIT_TEST 配下）からも参照するもの（Can_TxErrCount）。
#   上記以外（アルゴリズム定数等）はブロックスコープへ移動して解消する
#   （例: Crypto_Aes128_Rcon）。
